  },
  "memoria": {
    "total_mb": <inteiro>,
    "usada_mb": <inteiro>,
    "livre_mb": <inteiro>,
    "disponivel_mb": <inteiro>
  },
  "gpus": [
    {
      "nome": "<string>",
      "uso_pct": <inteiro>,
      "temperatura_c": <inteiro>,
      "vram_usada_mib": <inteiro>,
      "vram_livre_mib": <inteiro>,
      "vram_total_mib": <inteiro>
    }
  ]
}
