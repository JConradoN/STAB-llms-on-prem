#!/bin/bash
export LC_ALL=C

echo "=== SAÚDE DO SISTEMA ==="
echo "$(date '+%d/%m/%Y %H:%M:%S')"
echo ""

# CPU
CPU_USAGE=$(top -bn1 | awk '/^%Cpu/{gsub(",",".",$2); gsub(",",".",$4); printf "%.1f", $2+$4}')
CPU_TEMP=$(sensors 2>/dev/null | awk '/Package id 0:/{gsub(/[+°C]/,"",$4); print $4}')
echo "CPU"
echo "  Uso:         ${CPU_USAGE}%"
echo "  Temperatura: ${CPU_TEMP}°C"
echo ""

# Memória
read TOTAL USED FREE AVAIL <<< $(free -m | awk '/^Mem:/{print $2, $3, $4, $7}')
echo "MEMÓRIA"
echo "  Total:      ${TOTAL} MB"
echo "  Usada:      ${USED} MB"
echo "  Livre:      ${FREE} MB"
echo "  Disponível: ${AVAIL} MB"
echo ""

# GPU
GPU_INFO=$(nvidia-smi --query-gpu=name,utilization.gpu,temperature.gpu,memory.used,memory.free,memory.total --format=csv,noheader 2>/dev/null)
if [ -n "$GPU_INFO" ]; then
  while IFS=',' read -r NAME GPU_UTIL GPU_TEMP MEM_USED MEM_FREE MEM_TOTAL; do
    NAME=$(echo "$NAME" | xargs)
    echo "GPU: $NAME"
    echo "  Uso:        $(echo $GPU_UTIL | xargs)"
    echo "  Temp:       $(echo $GPU_TEMP | xargs)°C"
    echo "  VRAM usada: $(echo $MEM_USED | xargs) / $(echo $MEM_TOTAL | xargs)"
    echo "  VRAM livre: $(echo $MEM_FREE | xargs)"
  done <<< "$GPU_INFO"
else
  echo "GPU: nvidia-smi não disponível"
fi
echo ""

# Top 3 por CPU
CORES=$(nproc)
echo "TOP 3 PROCESSOS (CPU)"
ps aux --sort=-%cpu | awk -v cores="$CORES" 'NR>1 && NR<=4 {printf "  %-10s %5.1f%%  %s\n", $1, $3/cores, $11}'
echo ""

# Top 3 por memória
echo "TOP 3 PROCESSOS (MEMÓRIA)"
ps aux --sort=-%mem | awk 'NR>1 && NR<=4 {printf "  %-10s %5s%%  %s\n", $1, $4, $11}'
